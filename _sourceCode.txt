# ------------------------------------------------------------

# ------------------------------------------------------------
# 5. CONDICIONAL IF / ELIF / ELSE
# ------------------------------------------------------------

if n < 0:
    print("negativo")
elif n == 0:
    print("zero")
elif n < 10:
    print("pequeno")
else:
    print("grande")

# ------------------------------------------------------------

# ------------------------------------------------------------

# ------------------------------------------------------------
# 7. CICLO FOR COM RANGE
# ------------------------------------------------------------

for i in range(2, 8):

    print(i)

# ------------------------------------------------------------
# 8. FOR EM LISTA

# ------------------------------------------------------------
# 10. FUNCOES COM RETORNO
# ------------------------------------------------------------

    return a + b

# ------------------------------------------------------------
# 11. FUNCOES COM ANOTACOES DE TIPO
# ------------------------------------------------------------

    return a * b

print(r)

# ------------------------------------------------------------
# 12. FUNCOES RECURSIVAS
# ------------------------------------------------------------
def fatorial(n):
    if n <= 1:
        return 1
    return n * fatorial(n - 1)

print(fatorial(5))
print(fatorial(0))

# ------------------------------------------------------------
# 13. FUNCOES BUILTIN
# ------------------------------------------------------------

print(len("hello"))
print(range(3))

# ------------------------------------------------------------
# 14. OPERADORES LOGICOS EM QUERY
# ------------------------------------------------------------
a = 5
b = 10
if a < b and b < 20:
    print("ambos verdadeiros")

if a > b or b == 10:
    print("pelo menos um verdadeiro")

if not a == b:
    print("diferentes")

# ------------------------------------------------------------

# ------------------------------------------------------------

# ------------------------------------------------------------
# 18. WHILE COM CONDICAO COMPOSTA
# ------------------------------------------------------------
